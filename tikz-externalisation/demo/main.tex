% One document, one figure file, read twice: once plain, once with \ifRed on.
% \fig is the experiment: fixed, off, chain, override, edef, or inline.
\documentclass{article}
\pdfcompresslevel=0  % uncompressed, so the colour can be read with grep
\usepackage{tikz}
\usetikzlibrary{external}
\tikzexternalize[prefix=figs/\fig-]
\newif\ifRed
\begin{document}
\input{\fig-circle}%            plain: a blue circle
{\Redtrue\input{\fig-circle}}%  flag on: a red circle
\end{document}
